async function subscribeFCM(userName) { try { if (!("Notification" in window) || !("serviceWorker" in navigator) || !("PushManager" in window)) return; let permission = Notification.permission; // 如果從未授權過，嘗試要求授權 if (permission === "default") { permission = await Notification.requestPermission(); } if (permission !== "granted") return; const swReg = await navigator.serviceWorker.register("/firebase-messaging-sw.js"); await navigator.serviceWorker.ready; console.log("SW ready ✅"); let subscription = await swReg.pushManager.getSubscription(); // 🚨 核心修改：如果已經有訂閱了，千萬「不要」取消再重新訂閱，直接用舊的！ // 否則自動登入時因為沒有點擊按鈕，重新申請會被瀏覽器安全機制擋下。 if (!subscription) { const convertedKey = urlBase64ToUint8Array(FCM_VAPID_KEY); subscription = await swReg.pushManager.subscribe({ userVisibleOnly: true, applicationServerKey: convertedKey }); console.log("✨ 成功建立全新訂閱"); } else { console.log("🔄 使用已存在的推播訂閱"); } const subJson = JSON.stringify(subscription.toJSON()); const membersSnap = await getDocs(query(collection(db, "members"), where("姓名", "==", userName))); for (const d of membersSnap.docs) { const currentData = d.data(); // 只有當資料庫裡的訂閱碼跟現在不一樣或空缺時，才去更新資料庫 (減少不必要的讀寫) if (currentData.pushSubscription !== subJson) { await updateDoc(doc(db, "members", d.id), { pushSubscription: subJson }); console.log("訂閱已更新至 Firestore"); } } } catch(e) { console.warn("推播訂閱失敗:", e.message); } }
